# ===== Git Bash 环境变量 =====
export JAVA_HOME="/c/Users/16560/.android-sdk-tools/jdk-17.0.20.1+1"
export ANDROID_HOME="/c/Users/16560/.android-sdk-tools/android-sdk"
export ANDROID_SDK_ROOT="$ANDROID_HOME"
export PATH="$JAVA_HOME/bin:$ANDROID_HOME/platform-tools:$ANDROID_HOME/cmdline-tools/latest/bin:$PATH"

# ===== 编译命令 =====
# 进入 android 目录：cd client/android
# 编译 debug APK（调试签名，可直接安装）：
#   ./gradlew.bat assembleDebug --no-daemon
# 输出：client/android/app/build/outputs/apk/debug/app-debug.apk

# 编译 release APK（需正式签名 keystore，上架市场用）：
#   ./gradlew.bat assembleRelease --no-daemon

# ===== 网络镜像备忘 =====
# - JDK：清华镜像 https://mirrors.tuna.tsinghua.edu.cn/Adoptium/17/jdk/x64/windows/
# - Android SDK：Google 官方 dl.google.com（本机直连可达）
# - Gradle 发行版：services.gradle.org（本机直连可达）
# - curl 下载大文件需加 --ssl-no-revoke（Windows schannel 证书吊销检查问题）
